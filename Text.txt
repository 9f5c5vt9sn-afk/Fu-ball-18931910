const isFinished = (sc) => /^\d+:\d+$/.test(sc);

// ============ UI ============

function FormDots({ form }) {
  if (!form) return null;
  const colors = { S: "#16a34a", U: "#d97706", V: "#dc2626", C: "#94a3b8" };
  return (
    <div className="flex gap-1">
      {form.split("").map((c, i) => (
        <span key={i} className="inline-flex items-center justify-center w-5 h-5 rounded-full text-[10px] font-bold text-white"
          style={{ background: colors[c] || "#94a3b8" }}>{c}</span>
      ))}
    </div>
  );
}

function MatchRow({ m, club, accent }) {
  const AC = accent || { primary: "#6b4a2e", tint: "#efe6d8" };
  const isClub = (t) => t.includes(club);
  const done = isFinished(m[2]);
  const clubHome = isClub(m[1]);
  const win = done && ((clubHome && +m[2][0] > +m[2][2]) || (!clubHome && +m[2][2] > +m[2][0]));
  const draw = done && m[2][0] === m[2][2];
  return (
    <div className="flex items-center gap-3 px-3 py-2.5 rounded-xl border border-slate-200 bg-white"
      style={isClub(m[1]) || isClub(m[3]) ? { borderColor: AC.primary, background: AC.tint } : undefined}>
      <span className="text-xs text-slate-500 w-12 shrink-0 tabular-nums">{m[0]}</span>
      <span className={`flex-1 text-right text-sm font-medium truncate ${clubHome ? "font-bold" : "text-slate-700"}`}
        style={clubHome ? { color: AC.primary } : undefined}>{m[1]}</span>
      <span className={`shrink-0 w-16 text-center px-2 py-1 rounded-lg text-sm font-bold tabular-nums ${done ? "bg-slate-900 text-white" : "bg-slate-100 text-slate-600"}`}>
        {done ? m[2] : m[2] === "–" ? "tbd" : m[2]}
      </span>
      <span className={`flex-1 text-left text-sm font-medium truncate ${isClub(m[3]) ? "font-bold" : "text-slate-700"}`}
        style={isClub(m[3]) ? { color: AC.primary } : undefined}>{m[3]}</span>
    </div>
  );
}

function SpieltagView({ league, current, club, matchdays, clubOnly, accent }) {
  const AC = accent || { primary: "#6b4a2e", tint: "#efe6d8" };
  const [day, setDay] = useState(current);
  const games = matchdays[day] || [];
  const arr = Array.isArray(games[0]) && Array.isArray(games[0][0]) ? games[0] : games;
  return (
    <div>
      <div className="flex items-center justify-between mb-4 gap-2">
        <button onClick={() => setDay(Math.max(1, day - 1))} disabled={day <= 1}
          className="px-4 py-2 rounded-xl text-white text-sm font-semibold disabled:opacity-30 transition hover:opacity-80"
          style={{ background: AC.primary }}>←</button>
        <div className="flex items-center gap-2">
          <select value={day} onChange={(e) => setDay(+e.target.value)}
            className="px-4 py-2 rounded-xl border bg-white text-sm font-semibold focus:outline-none"
            style={{ borderColor: AC.primary }}>
            {Object.keys(matchdays).map((d) => (
              <option key={d} value={d}>{day === +d ? "▶ " : ""}{d}. Spieltag{+d === current ? " (aktuell)" : ""}</option>
            ))}
          </select>
        </div>
        <button onClick={() => setDay(Math.min(34, day + 1))} disabled={day >= 34}
          className="px-4 py-2 rounded-xl text-white text-sm font-semibold disabled:opacity-30 transition hover:opacity-80"
          style={{ background: AC.primary }}>→</button>
      </div>
      <div className="flex gap-2 mb-4 text-xs">
        {day === current && <span className="px-2 py-1 rounded-full text-white font-semibold" style={{ background: AC.primary }}>Aktueller Spieltag</span>}
        {day === current - 1 && <span className="px-2 py-1 rounded-full bg-slate-600 text-white font-semibold">Vorangegangener Spieltag</span>}
        {day === current + 1 && <span className="px-2 py-1 rounded-full bg-sky-600 text-white font-semibold">Kommender Spieltag</span>}
        {day < current - 1 && <span className="px-2 py-1 rounded-full bg-slate-500 text-white font-semibold">Abgeschlossen</span>}
        {day > current + 1 && <span className="px-2 py-1 rounded-full bg-slate-400 text-white font-semibold">Ausstehend</span>}
      </div>
      <div className="space-y-2">
        {arr.length === 0 ? (
          <p className="text-sm text-slate-500 py-6 text-center">Keine Daten für diesen Spieltag hinterlegt.</p>
        ) : arr.map((m, i) => <MatchRow key={i} m={m} club={club} accent={accent} />)}
        {clubOnly && arr.length === 1 && (
          <p className="text-[11px] text-slate-400 text-center pt-1">Für diesen Spieltag ist das {club}-Spiel hinterlegt.</p>
        )}
      </div>
    </div>
  );
}

function TableView({ data, club, league, accent }) {
  const AC = accent || { primary: "#6b4a2e", tint: "#efe6d8" };
  return (
    <div className="overflow-x-auto rounded-2xl border border-slate-200">
      <table className="w-full text-sm">
        <thead>
          <tr className="bg-slate-100 text-slate-600 text-xs uppercase tracking-wide">
            <th className="py-2.5 pl-3 pr-1 text-left w-8">#</th>
            <th className="py-2.5 text-left">Verein</th>
            <th className="py-2.5 px-1 text-center w-8">Sp</th>
            <th className="py-2.5 px-1 text-center w-8 hidden sm:table-cell">S</th>
            <th className="py-2.5 px-1 text-center w-8 hidden sm:table-cell">U</th>
            <th className="py-2.5 px-1 text-center w-8 hidden sm:table-cell">V</th>
            <th className="py-2.5 px-1 text-center w-16">Tore</th>
            <th className="py-2.5 px-1 text-center w-10 hidden sm:table-cell">Diff</th>
            <th className="py-2.5 px-1 text-center w-10">Pkt</th>
            {league === "bl2" && <th className="py-2.5 pr-3 text-center hidden md:table-cell">Form</th>}
          </tr>
        </thead>
        <tbody>
          {data.map((r) => {
            const isClub = r.team.includes(club) || (club.includes("Altona") && r.team.includes("Altona"));
            const zone = league === "bl2"
              ? (r.pl <= 2 ? "border-l-4 border-emerald-500" : r.pl === 3 ? "border-l-4 border-sky-500" : r.pl >= 16 ? "border-l-4 border-red-400" : "border-l-4 border-transparent")
              : (r.pl === 1 ? "border-l-4 border-emerald-500" : r.pl >= 16 ? "border-l-4 border-red-400" : "border-l-4 border-transparent");
            return (
              <tr key={r.pl} className={`${isClub ? "font-bold" : r.pl % 2 ? "bg-white" : "bg-slate-50/60"} ${zone}`} style={isClub ? { background: AC.tint, color: AC.primary } : undefined}>
                <td className="py-2 pl-3 pr-1 text-slate-500 tabular-nums">{r.pl}</td>
                <td className="py-2">{r.team}</td>
                <td className="py-2 px-1 text-center tabular-nums">{r.sp}</td>
                <td className="py-2 px-1 text-center tabular-nums hidden sm:table-cell">{r.s}</td>
                <td className="py-2 px-1 text-center tabular-nums hidden sm:table-cell">{r.u}</td>
                <td className="py-2 px-1 text-center tabular-nums hidden sm:table-cell">{r.v}</td>
                <td className="py-2 px-1 text-center tabular-nums text-slate-600">{r.tore}</td>
                <td className="py-2 px-1 text-center tabular-nums hidden sm:table-cell">{r.diff > 0 ? "+" + r.diff : r.diff}</td>
                <td className="py-2 px-1 text-center font-bold tabular-nums">{r.pkt}</td>
                {league === "bl2" && <td className="py-2 pr-3 hidden md:table-cell"><FormDots form={r.form} /></td>}
              </tr>
            );
          })}
        </tbody>
      </table>
      <div className="flex flex-wrap gap-4 px-3 py-2 bg-slate-100 text-[11px] text-slate-600 border-t">
        {league === "bl2" ? (
          <>
            <span><span className="inline-block w-2.5 h-2.5 bg-emerald-500 rounded-sm align-middle mr-1" />Aufstieg</span>
            <span><span className="inline-block w-2.5 h-2.5 bg-sky-500 rounded-sm align-middle mr-1" />Relegation</span>
            <span><span className="inline-block w-2.5 h-2.5 bg-red-400 rounded-sm align-middle mr-1" />Abstieg</span>
          </>
        ) : (
          <>
            <span><span className="inline-block w-2.5 h-2.5 bg-emerald-500 rounded-sm align-middle mr-1" />Aufstiegsrunde Regionalliga</span>
            <span><span className="inline-block w-2.5 h-2.5 bg-red-400 rounded-sm align-middle mr-1" />Abstiegszone</span>
          </>
        )}
      </div>
    </div>
  );
}

function ClubPage({ club, leagueName, table, league, current, matchdays, clubOnly, nextGame, pos, clubPts, accent }) {
  const AC = accent || { primary: "#6b4a2e", tint: "#efe6d8" };
  const [view, setView] = useState("spieltag");
  return (
    <div>
      <div className="flex flex-wrap items-center justify-between gap-3 mb-5">
        <div>
          <h2 className="text-2xl font-extrabold text-slate-900">{club}</h2>
          <p className="text-sm text-slate-500">{leagueName} · Saison 2026/27 · {pos}. Platz · {clubPts != null ? clubPts : "–"} Punkte</p>
        </div>
        {nextGame && (
          <div className="px-4 py-2 rounded-xl text-white text-sm" style={{ background: AC.primary }}>
            <span className="opacity-75 mr-2">Nächstes Spiel:</span>
            <strong>{nextGame[1]} – {nextGame[3]}</strong> ({nextGame[0]})
          </div>
        )}
      </div>
      <div className="flex gap-2 mb-5">
        {[["spieltag", "🗓 Spieltage"], ["tabelle", "📊 Tabelle"]].map(([k, label]) => (
          <button key={k} onClick={() => setView(k)}
            className={`px-4 py-2 rounded-xl text-sm font-semibold transition ${view === k ? "text-white shadow" : "text-slate-600 hover:opacity-80"}`}
            style={view === k ? { background: AC.primary } : { background: AC.tint, color: AC.primary }}>
            {label}
          </button>
        ))}
      </div>
      {view === "tabelle"
        ? <TableView data={table} club={club} league={league} accent={accent} />
        : <SpieltagView current={current} club={club} matchdays={matchdays} clubOnly={clubOnly} accent={accent} />}
    </div>
  );
}
